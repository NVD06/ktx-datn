SELECT i.code AS hoa_don, i.status AS trang_thai_hd,
       left(p."txnRef", 8) || '…' AS ma_gd,
       p.status AS trang_thai_gd, p."vnpResponseCode" AS ma_phan_hoi,
       '…' || right(p."vnpTransactionNo", 4) AS ma_gd_vnpay,
       p.amount AS so_tien,
       to_char(p."createdAt" AT TIME ZONE 'UTC' AT TIME ZONE 'Asia/Ho_Chi_Minh', 'DD/MM/YYYY HH24:MI:SS') AS tao_luc,
       to_char(p."paidAt" AT TIME ZONE 'UTC' AT TIME ZONE 'Asia/Ho_Chi_Minh', 'DD/MM/YYYY HH24:MI:SS') AS tra_luc
FROM "Payment" p JOIN "Invoice" i ON i.id = p."invoiceId"
WHERE i.code = 'INV-2026-0010'
ORDER BY p."createdAt"
